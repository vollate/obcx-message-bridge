include(GoogleTest)
set(_bridge_test_labels "unit\\;actor-package\\;bridge")

function(add_bridge_check name)
  add_executable(${name} ${ARGN})
  obcx_package_target(${name} ROLE test)
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
  target_link_libraries(${name} PRIVATE bridge_core)
  if(name STREQUAL "image_url_validator_test")
    gtest_discover_tests(${name} DISCOVERY_MODE PRE_TEST PROPERTIES LABELS "${_bridge_test_labels}" TIMEOUT 30)
  else()
    gtest_discover_tests(${name} DISCOVERY_MODE PRE_TEST PROPERTIES LABELS "${_bridge_test_labels}")
  endif()
endfunction()
